cmake_minimum_required(VERSION 3.24)
project(nightseam_consumer LANGUAGES CXX)

if(NOT NIGHTSEAM_SOURCE)
  message(FATAL_ERROR "NIGHTSEAM_SOURCE must name the copied source package")
endif()
add_subdirectory("${NIGHTSEAM_SOURCE}" nightseam)
if(TARGET nightseam_cpp_testee OR TARGET nightseam_peer_tests)
  message(FATAL_ERROR "A source consumer must not build private testees or component tests by default")
endif()
add_executable(nightseam_consumer main.cpp)
target_link_libraries(nightseam_consumer PRIVATE Nightseam::runtime)
enable_testing()
add_test(NAME consumer.roundtrip COMMAND nightseam_consumer)
set_tests_properties(consumer.roundtrip PROPERTIES TIMEOUT 15)
