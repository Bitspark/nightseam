cmake_minimum_required(VERSION 3.24)
project(nightseam LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(NIGHTSEAM_BUILD_TESTS "Build the C++ component invariant tests" ${PROJECT_IS_TOP_LEVEL})
if(NIGHTSEAM_BUILD_TESTS)
  include(CTest)
endif()
include(cmake/dependencies.cmake)
find_package(Threads REQUIRED)
add_subdirectory(duplex/cpp)
add_subdirectory(runtime/cpp)

option(NIGHTSEAM_BUILD_CONFORMANCE "Build the private JSON-lines conformance driver" ${PROJECT_IS_TOP_LEVEL})
if(NIGHTSEAM_BUILD_CONFORMANCE)
  add_subdirectory(conformance/cpp)
endif()
